             HuggingFaceTokenizer tok = HuggingFaceTokenizer.builder().optTokenizerPath(Paths.get(dir + "/tokenizer.json")).optTruncation(false).optPadding(false).build()) {
          for (NodeInfo n : s.getInputInfo().values()) System.out.println("input " + n.getName() + " " + n.getInfo());

          for (String p : ps) {
            Encoding e = tok.encode(q, p);
            long[] ids = e.getIds(); long[] types = e.getTypeIds(); long[] mask = e.getAttentionMask();

          }
          // padded batch of 2 and a 20x512 timing

            in.put("input_ids", OnnxTensor.createTensor(env, ids));
            in.put("attention_mask", OnnxTensor.createTensor(env, mask));

          }
        }

          }
